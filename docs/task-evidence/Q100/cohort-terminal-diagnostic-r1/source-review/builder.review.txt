class BuildInputError(ValueError):
    def __init__(self, code):
        super().__init__(code)
        self.code = code


def _need(condition, code):
    if not condition:
        raise BuildInputError(code)


def _sha(raw):
    return hashlib.sha256(raw).hexdigest()


def _is_reparse(info):
    return stat.S_ISLNK(info.st_mode) or bool(
        getattr(info, "st_file_attributes", 0) & 0x400)


def _inside(path, root, code):
    try:
        path.relative_to(root)
    except ValueError:
        raise BuildInputError(code) from None


def _bootstrap_read(path, expected_path, limit, expected_sha, code):
    _need(str(path) == str(expected_path), code + "_EXPECTED")
    try:
        resolved = path.resolve(strict=True)
        info = path.lstat()
    except (OSError, RuntimeError, ValueError):
        raise BuildInputError(code + "_PATH") from None
    _need(path.is_absolute() and str(resolved) == str(path)
          and stat.S_ISREG(info.st_mode) and not _is_reparse(info), code + "_IDENTITY")
    for parent in (path, *path.parents):
        try:
            current = parent.lstat()
        except OSError:
            raise BuildInputError(code + "_COMPONENT") from None
        _need(not _is_reparse(current), code + "_REPARSE")
    try:
        raw = path.read_bytes()
    except OSError:
        raise BuildInputError(code + "_READ") from None
    _need(0 < len(raw) <= limit and _sha(raw) == expected_sha, code + "_HASH")
    return raw

def _check_allowed(path, code):
    allowed = (D_ROOT, N_ROOT, E_ROOT, F_ROOT, T_ROOT, U_ROOT)
    if any(path == root or root in path.parents for root in allowed):
        return
    anchors = {
        H_HARNESS, PYTHON_TOOL, PS5_TOOL, E_VERIFIER, BASE_OVERLAY,
    }
    _need(path in anchors, code)


def _read_ref(n_builder, role, path, expected_bytes, expected_sha, limit, code):
    canonical = n_builder._canonical(str(path), code + "_PATH")
    _check_allowed(canonical, code + "_ALLOWLIST")
    raw = n_builder._read_file(canonical, limit, code)
    _need(len(raw) == expected_bytes and _sha(raw) == expected_sha, code + "_PIN")
    return {"role": role, "path": str(canonical), "bytes": len(raw),
            "sha256": _sha(raw)}, raw

def _directory(n_builder, path, expected_names, code):

    _check_allowed(canonical, code + "_ALLOWLIST")
    n_builder._lstat_directory(canonical, code)
    names = n_builder._directory_names(canonical, code)
    _need(len(names) == len(expected_names) and set(names) == set(expected_names),
          code + "_CLOSURE")

def _preflight_evidence(n_builder):

    # validation. This is read-only and does not claim semantic qualification.
    for name, value in globals().items():
        if "SHA256" not in name:
            continue
        if type(value) is str:
            _need(HEX.fullmatch(value) is not None, "PIN_FORMAT_" + name)
        elif type(value) is dict:
            for key, digest in value.items():
                _need(type(key) is str and HEX.fullmatch(digest) is not None,
                      "PIN_FORMAT_" + name)

    rows, raw_by_role = _pinned_rows(n_builder)
    evidence, evidence_raw = _qualification_refs(n_builder)
    _observation, observation_raw = _observation_ref(n_builder)
    providers = _provider_refs(n_builder)
    tools = _tool_refs(n_builder)
    contracts = _contract_refs(n_builder)
    _need(len(rows) == 9 and len(evidence) == 7 and len(providers) == 5
          and len(tools) == 2 and len(contracts) == 3,
          "PREFLIGHT_REFERENCE_COUNTS")
    projection = n_builder._json_load(observation_raw, 128 * 1024,
                                      "OBSERVATION_PROJECTION")
    summary = n_builder._json_load(evidence_raw["stdout.txt"], 1024,
                                   "QUALIFICATION_STDOUT")
    _need(type(projection) is dict and type(summary) is dict
          and projection.get("status") == "PASS_PRIOR_INPUT_ONLY"
          and projection.get("projectionIsFrozenDescriptor") is False
          and projection.get("qualificationSummary") == summary,
          "PREFLIGHT_OBSERVATION_BINDING")
    q_hashes = projection.get("qualificationReceiptHashes")
    _need(type(q_hashes) is dict
          and q_hashes == D_QUALIFICATION_SHA256,
          "PREFLIGHT_QUALIFICATION_RECEIPTS")
    expected_artifacts = [
        ("native-observe-r3/owner.json", 547,
         "1559355863aae9b680eeab3334c35953b4d239f42ab774b83173053b25808e79"),

         "8ac7a7ffc92dfb316be50a9da490d5ad51c494dd986fb18b7e821ca27e32c857"),

         "29a725862bd01151b67ef3d839de066b3e05d845f1744acf2153a51754ca244a"),
        ("native-observe-r3/result.json", 178,
         "c2dd1f6267219e8055cd043f803c1c836fcad7e51849de8862693fba5e089a7c"),
        ("native-observe-r3/stderr.txt", 0,
         "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855"),
        ("native-observe-r3/stdout.txt", 328,
         "42dacdb5c346b4595a2d4d07436723f4be8206ff778d31a233906318d484d0a5"),
        ("native-data-r3/op-1.jsonl", 377138,
         "cc2463e37c9636f1407434fd2f0d69fda660699e4899059273d1b72a71e02e58"),
        ("native-data-r3/result.json", 46323,
         "cd44ad1c0db81a8f05d8620f49b78a9e06c2274e6739066b438607f532cf1431"),

    actual_artifacts = []
    for row in rows:
        rel = Path(row["path"]).relative_to(N_ROOT).as_posix()
        actual_artifacts.append((rel, row["bytes"], row["sha256"]))
    _need(actual_artifacts == expected_artifacts, "PREFLIGHT_ARTIFACT_PINS")

          "N_READER_HOST_API")

        try:
            return reader.host_document(raw_by_name[name], 512 * 1024)
        except Exception:
            raise BuildInputError("QUALIFICATION_HOST_" + name.upper()) from None

    owner = host("owner.json")
    parent = host("parent.json")
    process = host("process.json")
    request = host("request.json")

    _need(type(owner) is dict and type(parent) is dict and type(process) is dict
          and type(request) is dict and type(wrapper_result) is dict,
          "QUALIFICATION_RECEIPT_SHAPE")

          and process.get("ExitCode") == 0 and process.get("TerminationProven") is True

          and owner.get("request") == str(D_ROOT / "prior-input-qualify-r2" / "request.json")
          and owner.get("executable") == str(PS5_TOOL),
          "QUALIFICATION_OWNER_BINDING")

          and request.get("Arguments") == process.get("Arguments")
          and request.get("LimitMs") == 180000,
          "QUALIFICATION_REQUEST_BINDING")

          and process.get("TerminationProven") is wrapper_result.get("terminationProven") is True

          and stdout_doc.get("status") == "PASS_OBSERVED_FAILURE_COHORT_DIAGNOSIS_ONLY"

          and stdout_doc.get("controlLayoutConnectivity") is None
          and stdout_doc.get("trialLayoutConnectivity") is None,
          "QUALIFICATION_READER_SUMMARY")
    _need(type(stdout_doc.get("layoutQualification")) is bool
          and type(stdout_doc.get("routingTreatmentSelected")) is bool
          and type(stdout_doc.get("productionAdopted")) is bool
          and type(stdout_doc.get("feasibilityClaimed")) is bool

def _contract_refs(n_builder):
    specs = [
        ("descriptor-contract", D_ROOT / "DESCRIPTOR_R2.md", 0, D_DESCRIPTOR_DOC_SHA256,
         256 * 1024),
        ("plan-contract", D_ROOT / "PLAN_R2.md", 0, D_PLAN_DOC_SHA256, 256 * 1024),
        ("wire-contract", D_WIRE_SOURCE, 0, D_WIRE_SHA256, MAX_WIRE_BYTES),
    ]
    result = []
    for role, path, _unused_size, digest, limit in specs:
        canonical = n_builder._canonical(str(path), "CONTRACT_PATH")
        _check_allowed(canonical, "CONTRACT_ALLOWLIST")
        raw = n_builder._read_file(canonical, limit, "CONTRACT_SOURCE")
        _need(_sha(raw) == digest, "CONTRACT_HASH_" + role.upper())
        result.append({"role": role, "path": str(canonical), "bytes": len(raw),
                       "sha256": _sha(raw)})
    return result


def _read_source(n_builder, path, limit, code):
    canonical = n_builder._canonical(str(path), code + "_PATH")
    raw = n_builder._read_file(canonical, limit, code)
    _need(raw.endswith(b"\n") and b"\r" not in raw
          and not raw.startswith(b"\xef\xbb\xbf"), code + "_FRAMING")
    try:
        raw.decode("utf-8", errors="strict")
    except UnicodeDecodeError:
        raise BuildInputError(code + "_UTF8") from None
    return canonical, raw

          and BASE_OVERLAY_SHA256 == "70cacaacc97cb3f01a1ad6221753e4d80a05479ba6663cd5dc6474a00dc4fb21",
          "OWNER_SOURCE_PINS")

    replacement_indexes = []
    replacement_rows = {}
    for ordinal, old_path, old_bytes, old_sha, code, new_path, new_role, new_bytes, new_sha in replacement_specs:
        old_row = n_sources[ordinal]
        _need(n_paths[ordinal] == old_path
              and old_row == {"path": old_path, "bytes": old_bytes, "sha256": old_sha},
              code + "_AUTHENTICATED_POSITION")
        _need(sum(1 for row in n_sources if row.get("path") == old_path

            base_raw = n_builder._read_file(base_path, MAX_SOURCE_BYTES, "BASE_OVERLAY")
            _need(len(base_raw) == old_bytes and _sha(base_raw) == old_sha,
                  "BASE_OVERLAY_AUTHENTICATION")
            owner_ref, _owner_raw = _actual_ref(
                n_builder, new_role, new_path, MAX_SOURCE_BYTES, "F_OWNER")
            _need(owner_ref["sha256"] == new_sha, "F_OWNER_HASH")
            replacement_rows[ordinal] = {
                "path": str(owner_ref["path"]),
                "bytes": owner_ref["bytes"],
                "sha256": owner_ref["sha256"],
            }
        else:
            canonical_new = n_builder._canonical(str(new_path), code + "_DERIVED_PATH")
            _check_allowed(canonical_new, code + "_DERIVED_ALLOWLIST")
            derived_raw = n_builder._read_file(canonical_new, MAX_SOURCE_BYTES,
                                                code + "_DERIVED_SOURCE")
            _need(len(derived_raw) == new_bytes and _sha(derived_raw) == new_sha,
                  code + "_DERIVED_PIN")
            replacement_rows[ordinal] = {
                "path": str(canonical_new), "bytes": len(derived_raw),
                "sha256": _sha(derived_raw),
            }
        replacement_indexes.append(ordinal)
    _need(replacement_indexes == [11, 1081, 1088]
          and len(set(replacement_indexes)) == 3,
          "SOURCE_REPLACEMENT_ORDINALS")
    study_path, study_raw = _read_source(n_builder, D_STUDY_SOURCE, MAX_SOURCE_BYTES,
                                         "D_STUDY_SOURCE")
    wire_path, wire_raw = _read_source(n_builder, D_WIRE_SOURCE, MAX_WIRE_BYTES,
                                       "D_WIRE_SOURCE")
    _need(_sha(wire_raw) == D_WIRE_SHA256, "D_WIRE_SOURCE_PIN")
    generated_path = n_builder._canonical(
        str(stage / "generated-source" / "com" / "lushprojects" / "circuitjs1" /
            "client" / "CohortTerminalTelemetryInput.java"),
        "GENERATED_SOURCE_PATH", require_exists=False,
    )
    source_rows = [dict(row) for row in n_sources]
    for ordinal in replacement_indexes:
        source_rows[ordinal] = replacement_rows[ordinal]
    _need(all(source_rows[index] == n_sources[index]
              for index in range(len(n_sources)) if index not in replacement_rows),
          "UNREPLACED_N_SOURCE_ROWS_CHANGED")
    source_rows.extend([
        {"path": str(study_path), "bytes": len(study_raw), "sha256": _sha(study_raw)},
        {"path": str(generated_path), "bytes": len(prior["generatedRaw"]),
         "sha256": _sha(prior["generatedRaw"])},
    ])
    paths = [row["path"] for row in source_rows]
    _need(len(source_rows) == 1092 and len(set(os.path.normcase(path) for path in paths)) == 1092,
          "D_SOURCE_COUNT_OR_DUPLICATE")
    for row in source_rows:
        _need(set(row) == {"path", "bytes", "sha256"}
              and type(row["bytes"]) is int and row["bytes"] > 0
              and HEX.fullmatch(row["sha256"]) is not None,
              "D_SOURCE_ROW")
    compiler_list = "".join('"' + path.replace("\\", "/") + '"\n'
                            for path in paths).encode("utf-8")
    _need(len(compiler_list) <= MAX_COMPILER_LIST_BYTES, "COMPILER_LIST_SIZE")
    return {
        "sources": source_rows,
        "sourcePaths": paths,
        "compilerSourceCount": len(source_rows),
        "compilerList": compiler_list,
        "wire": {"path": str(wire_path), "bytes": len(wire_raw),
                 "sha256": _sha(wire_raw)},
        "studyRaw": study_raw,
        "wireRaw": wire_raw,
        "generatedSourcePath": generated_path,
        "generatedSourceRaw": prior["generatedRaw"],
        "baseOverlayRowIndex": 11,
        "replacementOrdinals": replacement_indexes,
        "unchangedSourceRows": 1087,
    }


def _compile_request(stage, n_context):
    classes = stage / "classes"
    empty_sourcepath = stage / "empty-sourcepath"
    javac = n_context.get("javac")
    jars = n_context.get("jars")
    _need(type(javac) is dict and type(javac.get("path")) is str
          and type(jars) is dict
          and set(jars) == {"gwt-user-2.7.0.jar", "gwt-dev-2.7.0.jar"},
          "N_TOOL_SET")
    classpath = ";".join((
        str(classes), jars["gwt-user-2.7.0.jar"]["path"],
        jars["gwt-dev-2.7.0.jar"]["path"],
    ))
    arguments = [
        "-source", "7", "-target", "7", "-encoding", "UTF-8",
        "-classpath", classpath, "-sourcepath", str(empty_sourcepath),
        "-d", str(classes), "@" + str(stage / "compiler-sources.txt"),
    ]
    return {"FilePath": javac["path"], "Arguments": arguments,
            "LimitMs": COMPILE_LIMIT_MS}

def _canonical_stage(n_builder, stage_value, require_empty):
    stage = n_builder._canonical(stage_value, "STAGE_PATH", require_exists=False)
    _need(stage == STAGE_PATH and stage.parent == D_ROOT
          and STAGE_RE.fullmatch(stage.name) is not None, "STAGE_LOCATION")
    if require_empty:
        _need(not stage.exists(), "STAGE_ALREADY_EXISTS")
    else:
        n_builder._lstat_directory(stage, "STAGE_DIRECTORY")
    return stage

    n_builder._create_directory(stage, "STAGE_DIRECTORY")
    n_builder._create_directory(stage / "generated-source", "GENERATED_SOURCE_ROOT")
    package_dir = stage / "generated-source" / "com"
    n_builder._create_directory(package_dir, "GENERATED_PACKAGE_ROOT")
    package_dir = package_dir / "lushprojects"
    n_builder._create_directory(package_dir, "GENERATED_PACKAGE")
    package_dir = package_dir / "circuitjs1"
    n_builder._create_directory(package_dir, "GENERATED_PACKAGE")
    package_dir = package_dir / "client"
    n_builder._create_directory(package_dir, "GENERATED_PACKAGE")
    _need(package_dir / "CohortTerminalTelemetryInput.java"
          == identity["generatedSourcePath"], "GENERATED_LOCATION")
    n_builder._write_new(identity["generatedSourcePath"], identity["generatedSourceRaw"],
                         MAX_GENERATED_BYTES, "GENERATED_SOURCE")

    n_builder._write_new(stage / "source-manifest.json", manifest_raw,

        "generated-source", "classes", "empty-sourcepath", "compiler-sources.txt",
        "compiler-request.json", "source-manifest.json",
    }
    names = n_builder._directory_names(stage, "STAGE_DIRECTORY")
    _need(len(names) == len(expected) and set(names) == expected,
          "PREPARED_STAGE_CLOSURE")
    return manifest_raw

                         MAX_DESCRIPTOR_BYTES, "DESCRIPTOR")

def _validate_source_input(n_builder, n_context, stage, expected_sha, *, finalized=False):
    _need(type(expected_sha) is str and HEX.fullmatch(expected_sha) is not None,
          "SOURCE_MANIFEST_SHA_TYPE")
    _validate_stage_closure(n_builder, stage, finalized=finalized)
    n_builder._lstat_directory(stage / "classes", "CLASSES_ROOT")
    empty_sourcepath = n_builder._canonical(str(stage / "empty-sourcepath"),
                                            "EMPTY_SOURCEPATH")
    n_builder._lstat_directory(empty_sourcepath, "EMPTY_SOURCEPATH")
    _need(not n_builder._directory_names(empty_sourcepath, "EMPTY_SOURCEPATH"),
          "EMPTY_SOURCEPATH_NOT_EMPTY")
    prior = _prior_inputs(n_builder, n_context, write_descriptor=False)
    manifest, identity, request, request_raw = _manifest_identity(
        n_builder, n_context, prior, stage)
    source_path = n_builder._canonical(str(stage / "source-manifest.json"),
                                       "SOURCE_MANIFEST_PATH")
    source_raw = n_builder._read_file(source_path, MAX_MANIFEST_BYTES, "SOURCE_MANIFEST")
    _need(_sha(source_raw) == expected_sha
          and source_raw == n_builder._canonical_json(manifest),
          "SOURCE_MANIFEST_BINDING")
    compiler_list = n_builder._read_file(stage / "compiler-sources.txt",
                                         MAX_COMPILER_LIST_BYTES, "COMPILER_LIST")
    _need(compiler_list == identity["compilerList"], "COMPILER_LIST_BINDING")
    request_path = n_builder._canonical(str(stage / "compiler-request.json"),
                                        "COMPILER_REQUEST_PATH")
    request_disk = n_builder._read_file(request_path, MAX_SMALL_JSON_BYTES,
                                        "COMPILER_REQUEST")
    _need(request_disk == request_raw and request == n_builder._strict_document(
        request_disk, MAX_SMALL_JSON_BYTES, "COMPILER_REQUEST"),
        "COMPILER_REQUEST_BINDING")
    _generated_source_closure(n_builder, stage, identity["generatedSourcePath"])
    generated_disk = n_builder._read_file(identity["generatedSourcePath"],
                                          MAX_GENERATED_BYTES, "GENERATED_SOURCE")
    _need(generated_disk == identity["generatedSourceRaw"]
          and _sha(generated_disk) == manifest["generatedInputSha256"],
          "GENERATED_SOURCE_BINDING")
    return prior, manifest, identity, request, request_raw, source_raw


def _host_document(reader, n_builder, path, limit, code):
    canonical = n_builder._canonical(str(path), code + "_PATH")
    raw = n_builder._read_file(canonical, limit, code)
    try:
        value = reader.host_document(raw, limit)
    except Exception:
        raise BuildInputError(code + "_JSON") from None
    return canonical, raw, value


def _validate_compile_call(n_builder, n_context, call_value, stage, expected_request):
    call_dir = n_builder._canonical(call_value, "COMPILE_CALL_PATH")
    _need(call_dir == D_ROOT / COMPILE_CALL_NAME and call_dir.parent == D_ROOT
          and COMPILE_CALL_RE.fullmatch(call_dir.name) is not None,
          "COMPILE_CALL_LOCATION")
    _directory(n_builder, call_dir, [
        "owner.json", "parent.json", "process.json", "request.json",
        "result.json", "stderr.txt", "stdout.txt",
    ], "COMPILE_CALL_DIRECTORY")
    reader = n_context.get("baseReader")
    process_module = n_context.get("processModule")
    _need(reader is not None and callable(getattr(reader, "host_document", None))
          and callable(getattr(reader, "_validate_owner_call", None))
          and process_module is not None, "COMPILE_CALL_KERNELS")
    parent_path, parent_raw, parent = _host_document(
        reader, n_builder, call_dir / "parent.json", MAX_SMALL_JSON_BYTES, "COMPILE_PARENT")
    _request_path, request_raw, request = _host_document(
        reader, n_builder, call_dir / "request.json", MAX_SMALL_JSON_BYTES,
        "COMPILE_REQUEST_RECEIPT")
    _process_path, process_raw, process = _host_document(
        reader, n_builder, call_dir / "process.json", MAX_SMALL_JSON_BYTES,
        "COMPILE_PROCESS")
    owner_path, owner_raw, owner = _host_document(
        reader, n_builder, call_dir / "owner.json", MAX_SMALL_JSON_BYTES, "COMPILE_OWNER")
    result_path, result_raw, result = _host_document(
        reader, n_builder, call_dir / "result.json", MAX_SMALL_JSON_BYTES, "COMPILE_RESULT")
    _need(type(parent) is dict and type(process) is dict
          and type(request) is dict and type(owner) is dict and type(result) is dict,
          "COMPILE_CALL_SHAPE")
    _need(parent.get("schema") == "tsj-q100-reservation-provenance-parent-v1"
          and parent.get("process") == process and request == expected_request,
          "COMPILE_CALL_BINDING")
    _need(owner.get("schema") == "tsj-p9-root-call-owner-v1"
          and owner.get("request") == str(call_dir / "request.json"),
          "COMPILE_OWNER_BINDING")
    _need(process.get("ExitCode") == 0 and process.get("TerminationProven") is True
          and result.get("status") == "RETURNED" and result.get("exitCode") == 0
          and result.get("terminationProven") is True,
          "COMPILE_NOT_SUCCESS")
    stdout_raw = n_builder._read_file(call_dir / "stdout.txt", 2 * 1024 * 1024,
                                     "COMPILE_STDOUT")
    stderr_raw = n_builder._read_file(call_dir / "stderr.txt", 2 * 1024 * 1024,
                                     "COMPILE_STDERR")
    _need(process.get("Stdout", "").encode("utf-8") == stdout_raw
          and process.get("Stderr", "").encode("utf-8") == stderr_raw,
          "COMPILE_OUTPUT_BINDING")
    for key in ("ProcessRoot", "StdoutLog", "StderrLog"):
        value = process.get(key)
        _need(type(value) is str and Path(value).is_absolute()
              and not Path(value).exists(), "COMPILE_CLEANUP_" + key.upper())
    try:
        reader._validate_owner_call(
            call_dir, parent_path, parent, request, process,
            expected_request["FilePath"], expected_request["Arguments"],
            COMPILE_LIMIT_MS, process_module,
            required_stdout="", expected_stderr=reader.COMPILE_STDERR,
        )
    except Exception as failure:
        code = getattr(failure, "code", None)
        if type(code) is str and re.fullmatch(r"[A-Z0-9_]{1,96}", code):
            raise BuildInputError("COMPILE_OWNER_" + code) from None
        raise BuildInputError("COMPILE_OWNER_VALIDATION") from None
    return {
        "directory": call_dir,
        "parentPath": parent_path,
        "parentRaw": parent_raw,
        "parent": parent,
        "processRaw": process_raw,
        "process": process,
        "requestRaw": request_raw,
        "request": request,
        "ownerPath": owner_path,
        "ownerRaw": owner_raw,
        "resultPath": result_path,
        "resultRaw": result_raw,
    }


def _inventory_classes(n_builder, stage, source_manifest_sha):
    classes = n_builder._canonical(str(stage / "classes"), "CLASSES_ROOT")
    try:
        inventory, inventory_raw = n_builder._inventory_class_tree(
            classes, source_manifest_sha)
    except n_builder.BuildInputError as failure:
        code = getattr(failure, "code", None)
        if type(code) is str and re.fullmatch(r"[A-Z0-9_]{1,96}", code):
            raise BuildInputError("D_CLASS_" + code) from None
        raise BuildInputError("D_CLASS_INVENTORY") from None
    paths = {row["path"] for row in inventory["entries"]}
    required = {
        "com/lushprojects/circuitjs1/client/CohortTerminalTelemetryStudy.class",
        "com/lushprojects/circuitjs1/client/CohortTerminalTelemetryInput.class",
    }
    _need(required.issubset(paths), "D_STUDY_CLASSES_MISSING")
    return inventory, inventory_raw

def finalize(stage_value, compile_call_directory, source_manifest_sha256):
    n_builder = _load_n_builder()
    stage = _canonical_stage(n_builder, stage_value, require_empty=False)
    n_context = _load_n_context(n_builder)  # One complete N validation for this entry.

        n_builder, n_context, stage, source_manifest_sha256)
    call = _validate_compile_call(n_builder, n_context, compile_call_directory,
                                  stage, request)
    inventory, inventory_raw = _inventory_classes(n_builder, stage,
                                                   _sha(source_raw))
    inventory_path = stage / "class-inventory.json"
    build = {
        "sourceManifestPath": str(stage / "source-manifest.json"),
        "sourceManifestSha256": _sha(source_raw),
        "compileCallDirectory": str(call["directory"]),
        "compileParentPath": str(call["parentPath"]),
        "compileParentSha256": _sha(call["parentRaw"]),
        "classInventoryPath": str(inventory_path),
        "classInventorySha256": _sha(inventory_raw),

        "classBytes": inventory["totalBytes"],
    }
    run_manifest = _expected_run_manifest(input_manifest, build)

    return {
        "schema": "tsj-cohort-terminal-finalized-r2-v2",
        "status": "FINALIZED_ONLY",
        "runManifestPath": str(stage / "run-manifest.json"),
        "runManifestSha256": _sha(run_raw),
        "sourceManifestSha256": _sha(source_raw),
        "descriptorSha256": prior["descriptorSha256"],
        "generatedInputSha256": _sha(identity["generatedSourceRaw"]),
        "compilerSourceCount": identity["compilerSourceCount"],
        "classCount": inventory["classCount"],
        "classBytes": inventory["totalBytes"],
    }


def _validated_context(n_builder, n_context, prior, run_path, run_raw, run_manifest,
                       input_manifest, identity, source_raw, inventory, call):
    # Keep both the authenticated receipt bytes and parser projection. Pad rows
    # used by the consumer's independent coordinate checks come from this parsed
    # op-1 snapshot; canonical pad-land identity still comes from priorResult.
    first_line = prior["priorOperationRaw"].split(b"\n", 1)[0]
    try:
        prior_header = n_builder._json_load(first_line, 64 * 1024, "PRIOR_OPERATION_HEADER")
    except BuildInputError:
        raise BuildInputError("PRIOR_OPERATION_HEADER") from None
    host_reader = n_context["baseReader"]
    prior_reader = prior["priorReaderModule"]
    parse_operation = getattr(host_reader, "_parse_operation1", None)
    process = prior["priorProcess"]
    _need(callable(parse_operation) and type(process) is dict
          and type(process.get("Arguments")) is list
          and len(process["Arguments"]) == 9
          and type(n_context.get("p07Reader")) is types.ModuleType
          and type(n_context.get("wire")) is dict
          and n_context["wire"].get("sha256") == prior_reader.WIRE_SHA256,
          "PRIOR_OPERATION_PARSER_API")
    try:
        parsed_operation = parse_operation(
            prior["priorOperationRaw"], prior["priorResult"]["operation1"],
            process["Arguments"][5], n_context["sourceSha"],
            prior_reader.WIRE_SHA256, n_context["p07Reader"],
        )
    except Exception:
        raise BuildInputError("PRIOR_OPERATION_PARSE") from None
    _need(type(parsed_operation) is dict and type(parsed_operation.get("pads")) is list,
          "PRIOR_OPERATION_PARSED_SHAPE")
    context = {
        "manifest": run_manifest,
        "sourceSha": _sha(run_raw),
        "path": run_path,
        "sourceManifest": input_manifest,
        "sourceManifestSha256": _sha(source_raw),
        "sourcePaths": [row["path"] for row in input_manifest["sources"]],
        "sources": input_manifest["sources"],
        "tools": input_manifest["tools"],
        "wire": input_manifest["wire"],
        "classInventory": inventory,
        "classesRoot": n_context.get("classesRoot"),
        "generatedSourcePath": identity["generatedSourcePath"],
        "generatedSourceRaw": identity["generatedSourceRaw"],
        "generatedSourceSha256": _sha(identity["generatedSourceRaw"]),
        "descriptor": prior["descriptor"],
        "descriptorRaw": prior["descriptorRaw"],
        "descriptorPath": str(DESCRIPTOR_PATH),
        "descriptorSha256": prior["descriptorSha256"],
        "priorResultRaw": prior["priorResultRaw"],
        "priorResult": prior["priorResult"],
        "priorOperation1Raw": prior["priorOperationRaw"],
        "priorOperation1Header": prior_header,
        "priorParsedOperation1": parsed_operation,
        "priorReaderSummary": prior["priorReaderSummary"],
        "nContext": n_context,
        "baseReader": prior_reader,
        "priorReaderModule": prior_reader,
        "hostReader": host_reader,
        "processModule": n_context.get("processModule"),
        "java": n_context.get("java"),
        "javac": n_context.get("javac"),
        "jars": n_context.get("jars"),
        "parentAuditor": n_context.get("parentAuditor"),
        "compileCall": call,
    }
    context["sourceManifestPath"] = run_path.parent / "source-manifest.json"
    context["classesRoot"] = run_path.parent / "classes"
    return context


def validate_run_manifest(path_value, expected_sha256):
    n_builder = _load_n_builder()
    run_path = n_builder._canonical(path_value, "RUN_MANIFEST_PATH")
    _need(run_path.name == "run-manifest.json"
          and run_path.parent == STAGE_PATH, "RUN_MANIFEST_LOCATION")
    stage = _canonical_stage(n_builder, run_path.parent, require_empty=False)
    run_raw = n_builder._read_file(run_path, MAX_RUN_MANIFEST_BYTES, "RUN_MANIFEST")
    _need(type(expected_sha256) is str and HEX.fullmatch(expected_sha256) is not None
          and _sha(run_raw) == expected_sha256, "RUN_MANIFEST_HASH")
    run_manifest = n_builder._strict_document(run_raw, MAX_RUN_MANIFEST_BYTES,
                                              "RUN_MANIFEST")
    _need(type(run_manifest) is dict
          and run_manifest.get("schema") == "tsj-cohort-terminal-source-r2-run-v2"
          and type(run_manifest.get("build")) is dict, "RUN_MANIFEST_SCHEMA")
    n_context = _load_n_context(n_builder)  # One complete N validation for this entry.
    build = run_manifest["build"]
    source_sha = build.get("sourceManifestSha256")
    prior, input_manifest, identity, request, request_raw, source_raw = _validate_source_input(
        n_builder, n_context, stage, source_sha, finalized=True)
    call = _validate_compile_call(n_builder, n_context,
                                  build.get("compileCallDirectory"), stage, request)
    inventory, inventory_raw = _inventory_classes(n_builder, stage, source_sha)
    inventory_path = n_builder._canonical(str(stage / "class-inventory.json"),
                                          "CLASS_INVENTORY_PATH")
    inventory_disk = n_builder._read_file(inventory_path, MAX_CLASS_INVENTORY_BYTES,
                                          "CLASS_INVENTORY")
    _need(inventory_disk == inventory_raw
          and _sha(inventory_disk) == build.get("classInventorySha256"),
          "CLASS_INVENTORY_BINDING")
    expected_build = {
        "sourceManifestPath": str(stage / "source-manifest.json"),
        "sourceManifestSha256": _sha(source_raw),
        "compileCallDirectory": str(call["directory"]),
        "compileParentPath": str(call["parentPath"]),
        "compileParentSha256": _sha(call["parentRaw"]),
        "classInventoryPath": str(inventory_path),
        "classInventorySha256": _sha(inventory_raw),
        "classCount": inventory["classCount"],
        "classBytes": inventory["totalBytes"],
    }
    _need(build == expected_build
          and run_raw == n_builder._canonical_json(
              _expected_run_manifest(input_manifest, expected_build)),
          "RUN_MANIFEST_IDENTITY")
    _validate_stage_closure(n_builder, stage, finalized=True)
    _need(n_builder._read_file(run_path, MAX_RUN_MANIFEST_BYTES,
                               "RUN_MANIFEST_RECHECK") == run_raw,
          "RUN_MANIFEST_CHANGED")
    context = _validated_context(n_builder, n_context, prior, run_path, run_raw,
                                 run_manifest, input_manifest, identity,
                                 source_raw, inventory, call)
    context["sourceManifestPath"] = stage / "source-manifest.json"
    return context


def audit(run_manifest_path, run_manifest_sha256):
    return validate_run_manifest(run_manifest_path, run_manifest_sha256)


def _emit(value):
    raw = json.dumps(value, ensure_ascii=True, sort_keys=True,
                     separators=(",", ":"), allow_nan=False).encode("ascii")
    sys.stdout.buffer.write(raw + b"\n")
    sys.stdout.buffer.flush()


def _audit_summary(context):
    return {
        "schema": "tsj-cohort-terminal-source-audit-r2-v2",
        "status": "SOURCE_AUDIT_ONLY",
        "sourceSha256": context["sourceSha"],
        "sourceRows": len(context["sourcePaths"]),
        "compilerSourceCount": context["manifest"]["compilerSourceCount"],
        "descriptorSha256": context["descriptorSha256"],
        "generatedInputSha256": context["generatedSourceSha256"],
        "classCount": context["classInventory"]["classCount"],
        "classBytes": context["classInventory"]["totalBytes"],
        "wireSha256": context["wire"]["sha256"],
    }


def main(argv=None):
    import argparse
    parser = argparse.ArgumentParser(prog="build_inputs_r2.py")
    commands = parser.add_subparsers(dest="command", required=True)
    prep = commands.add_parser("prepare")
    prep.add_argument("stage", nargs="?", default=str(STAGE_PATH))
    finish = commands.add_parser("finalize")
    finish.add_argument("stage")
    finish.add_argument("compile_call_directory")
    finish.add_argument("source_manifest_sha256")
    check = commands.add_parser("audit")
    check.add_argument("run_manifest")
    check.add_argument("run_manifest_sha256")
    args = parser.parse_args(argv)
    try:
        if args.command == "prepare":
            result = prepare(args.stage)
        elif args.command == "finalize":
            result = finalize(args.stage, args.compile_call_directory,
                              args.source_manifest_sha256)
        else:
            result = _audit_summary(audit(args.run_manifest, args.run_manifest_sha256))
        _emit(result)
        return 0
    except BuildInputError as failure:
        code = failure.code if re.fullmatch(r"[A-Z0-9_]{1,96}", failure.code) else "INVALID_ERROR_CODE"
        sys.stderr.write("COHORT_TERMINAL_BUILD_FAIL " + code + "\n")
        sys.stderr.flush()
        return 2
    except Exception as failure:
        code = getattr(failure, "code", None)
        if type(code) is str and re.fullmatch(r"[A-Z0-9_]{1,96}", code):
            sys.stderr.write("COHORT_TERMINAL_BUILD_FAIL N_" + code + "\n")
            sys.stderr.flush()
            return 2
        kind = re.sub(r"[^A-Za-z0-9_]", "_", type(failure).__name__).upper()[:48]
        sys.stderr.write("COHORT_TERMINAL_BUILD_FAIL INTERNAL_" + (kind or "ERROR") + "\n")
        sys.stderr.flush()
        return 2


if __name__ == "__main__":
    raise SystemExit(main())
