class DescriptorInputError(ValueError):
    def __init__(self, code):
        super().__init__(code)
        self.code = code


def _need(condition, code):
    if not condition:
        raise DescriptorInputError(code)


def _sha(raw):
    return hashlib.sha256(raw).hexdigest()


def _is_int(value, low=0, high=2**63 - 1):
    return type(value) is int and low <= value <= high


def _pairs(items):
    result = {}
    for key, value in items:
        if key in result:
            raise DescriptorInputError("JSON_DUPLICATE_KEY")
        result[key] = value
    return result


def _no_float(_text):
    raise DescriptorInputError("JSON_FLOAT")


def _no_constant(_text):
    raise DescriptorInputError("JSON_NONFINITE")


def canonical_json(value):
    try:
        return (json.dumps(value, ensure_ascii=True, sort_keys=True,
                           separators=(",", ":"), allow_nan=False) + "\n").encode("ascii")
    except (TypeError, ValueError, UnicodeError):
        raise DescriptorInputError("JSON_SERIALIZE") from None


def strict_document(raw, limit=MAX_DESCRIPTOR_BYTES):
    _need(type(raw) is bytes and 0 < len(raw) <= limit, "DOCUMENT_SIZE")
    _need(raw.endswith(b"\n") and not raw.endswith(b"\n\n")
          and b"\r" not in raw and not raw.startswith(b"\xef\xbb\xbf"),
          "DOCUMENT_FRAMING")
    try:
        value = json.loads(raw[:-1].decode("utf-8", errors="strict"),
                           object_pairs_hook=_pairs,
                           parse_float=_no_float, parse_constant=_no_constant)
    except DescriptorInputError:
        raise
    except (UnicodeDecodeError, json.JSONDecodeError, ValueError):
        raise DescriptorInputError("DOCUMENT_JSON") from None
    _need(raw == canonical_json(value), "DOCUMENT_CANONICAL")
    return value


def _exact(value, keys, code):
    _need(type(value) is dict and set(value) == keys, code)


def _ascii(value, maximum=256, allow_empty=False, code="STRING"):
    _need(type(value) is str and (allow_empty or len(value) > 0)
          and len(value) <= maximum
          and all(32 <= ord(character) <= 126 for character in value), code)
    return value


def _hash(value, code):
    _need(type(value) is str and HEX.fullmatch(value) is not None, code)


def _ref(value, code):
    _exact(value, REF_KEYS, code + "_KEYS")
    _ascii(value["role"], code=code + "_ROLE")
    path = _ascii(value["path"], maximum=256, code=code + "_PATH")
    _need(re.fullmatch(r"[A-Z]:\\[^/]+(?:\\[^/]+)*", path) is not None,
          code + "_ABSOLUTE_PATH")
    _need(_is_int(value["bytes"], 0, 2**31 - 1), code + "_BYTES")
    _hash(value["sha256"], code + "_SHA")
    return value


def _ref_rows(value, count, code, roles=None):
    _need(type(value) is list and len(value) == count, code + "_COUNT")
    seen_roles = set()
    seen_paths = set()
    for index, row in enumerate(value):
        _ref(row, code + "_ROW")
        _need(row["role"] not in seen_roles and row["path"] not in seen_paths,
              code + "_DUPLICATE")
        seen_roles.add(row["role"])
        seen_paths.add(row["path"])
        if roles is not None:
            _need(row["role"] == roles[index], code + "_ROLE_ORDER")


def _pinned_ref(value, role, path, byte_count, digest, code):
    _ref(value, code)
    _need(value == {
        "role": role,
        "path": str(path),
        "bytes": byte_count,
        "sha256": digest,
    }, code + "_PIN")

          and operation["pass"] == 2 and operation["failedNetId"] == "A_COIL_LOW"
          and operation["startPadId"] == "QA.C" and operation["rootPadId"] == "DA.A"
          and operation["reason"] == "NO_PATH" and operation["reachedNetId"] == "DA.A"
          and operation["receiptOrdinal"] == 307 and operation["groupOrdinal"] == 98
          and operation["lineageOrdinal"] == 77 and operation["operationBytes"] == 265894
          and operation["holeCount"] == 48
          and operation["prefixGeometrySha256"]
          == "6b49cfb2d42a83007824b15c8c7a7c89cdadf627fed9b6d9d966692af0323ddb",
          "PRIOR_OPERATION_FIXED")

    identity = value["sourceIdentity"]
    _exact(identity, SOURCE_IDENTITY_KEYS, "SOURCE_IDENTITY_KEYS")
    for key in ("geometryBeforeSha256", "geometryAfterSha256", "detachedGeometrySha256",
                "effectsBeforeSha256", "effectsAfterSha256"):
        _hash(identity[key], "SOURCE_IDENTITY_" + key.upper())
    _need(_is_int(identity["traceCount"], 0, 1000000)
          and _is_int(identity["holeCount"], 0, 1000000),
          "SOURCE_IDENTITY_COUNTS")
    pad = identity["padLandIdentity"]
    _exact(pad, PAD_IDENTITY_KEYS, "PAD_IDENTITY_KEYS")
    _need(pad["format"] == PAD_FORMAT and _is_int(pad["rowCount"], 246, 246),
          "PAD_IDENTITY_FIXED")
    _hash(pad["sha256"], "PAD_IDENTITY_SHA")
    _need(pad["sha256"] == "541543170f9d48428f41ae881c596384597543ec03453fc9f9ea8d14033275a0"
          and identity["traceCount"] == 0 and identity["holeCount"] == 0,
          "SOURCE_IDENTITY_FIXED")
    _validate_claims(value["claims"], "CLAIMS")

    # Enforce printable ASCII and the frozen per-string bound at every depth.
    pending = [value]
    while pending:
        item = pending.pop()
        if type(item) is str:
            _ascii(item, code="DESCRIPTOR_STRING")
        elif type(item) is dict:
            pending.extend(item.keys())
            pending.extend(item.values())
        elif type(item) is list:
            pending.extend(item)
        else:
            _need(item is None or type(item) in (bool, int), "DESCRIPTOR_VALUE_TYPE")
    return value

    # Keep the D observation file as a cross-check only; never use it as authority.

def derive_descriptor(*, observation_ref, evidence_files, outcome, reader_summary,
                      approved_prior_artifact_rows, prior_source_closure,
                      native_copy_allowance_source, native_shared_budget_source,
                      provider_source_refs, provider_tool_refs, prior_result,
                      prior_result_raw, prior_operation_raw, pad_identity):
    _need(type(prior_result) is dict and type(prior_result_raw) is bytes,
          "PRIOR_RESULT_INPUT")
    _need(type(prior_operation_raw) is bytes, "PRIOR_OPERATION_INPUT")
    _need(_sha(prior_result_raw) == reader_summary.get("resultSha256"),
          "PRIOR_RESULT_BINDING")
    _need(_sha(prior_operation_raw) == reader_summary.get("operation1Sha256"),
          "PRIOR_OPERATION_BINDING")
    result_operation = prior_result.get("operation1")
    result_trial = prior_result.get("trialDisposition")
    captured = prior_result.get("capturedOrder")
    source = prior_result.get("source")
    reference = prior_result.get("reference")
    control = prior_result.get("control")
    trial = prior_result.get("trial")
    _need(type(result_operation) is dict and type(result_trial) is dict
          and type(captured) is dict and type(source) is dict
          and type(reference) is dict and type(control) is dict
          and type(trial) is dict, "PRIOR_RESULT_SHAPE")
    control_outcome = control.get("outcome")
    trial_outcome = trial.get("outcome")
    _need(type(control_outcome) is dict and "disposition" in control_outcome
          and type(trial_outcome) is dict and "disposition" in trial_outcome,
          "PRIOR_RESULT_OUTCOME_SHAPE")
    _need(prior_result.get("schema") == "tsj-failure-cohort-routing-v1",
          "PRIOR_RESULT_SCHEMA")
    _need(type(prior_result.get("runId")) is str, "PRIOR_RESULT_RUN_ID")

    rows_by_role = {row["role"]: row for row in approved_prior_artifact_rows}
    _need(len(rows_by_role) == len(approved_prior_artifact_rows), "PRIOR_ROWS_DUPLICATE")
    _need(rows_by_role.get("data-result", {}).get("sha256") == _sha(prior_result_raw)
          and rows_by_role.get("data-op-1", {}).get("sha256") == _sha(prior_operation_raw),
          "PRIOR_ROW_BINDING")
    _need(result_operation.get("receiptSha256") == _sha(prior_operation_raw)
          and result_operation.get("receiptPath") == "op-1.jsonl",
          "PRIOR_OPERATION_RECEIPT")
    _need(reader_summary.get("manifestSha256") == prior_source_closure.get(
        "runManifestRef", {}).get("sha256")
          and reader_summary.get("resultSha256") == rows_by_role["data-result"]["sha256"]
          and reader_summary.get("operation1Sha256") == rows_by_role["data-op-1"]["sha256"],
          "QUALIFIED_SUMMARY_BINDING")
    _need(reader_summary.get("readerSha256")
          == next((row["sha256"] for row in provider_source_refs
                   if row.get("role") == "prior-reader-source"), None),
          "QUALIFIED_READER_BINDING")
    _need(type(result_trial) is dict and result_trial.get("status") == "RUN"
          and result_trial.get("reason") == "TRIAL_EXECUTED"
          and result_trial.get("feedbackValidated") is True,
          "PRIOR_TRIAL_DISPOSITION")
    _need(type(captured.get("owner")) is dict, "PRIOR_CAPTURE_OWNER")
    owner_raw = captured["owner"]
    owner = {
        "boardMode": owner_raw.get("policy"),
        "maxExpansions": owner_raw.get("maximumExpansions"),
        "expandedAtFailure": owner_raw.get("expandedAtFailure"),
        "maxVias": owner_raw.get("maximumVias"),
        "viaCost": owner_raw.get("viaCost"),
        "maxTransitions": owner_raw.get("policyTransitions"),
        "branchStrategy": owner_raw.get("branchOrderStrategy"),
        "netStrategy": owner_raw.get("netOrderStrategy"),
        "roleWidthsEnabled": owner_raw.get("roleWidths"),
        "orderingStreamEnabled": owner_raw.get("orderingStream"),
    }
    seeds_raw = {key: result_operation.get(key)
                 for key in ("rootSeed", "placementSeed", "routingSeed")}
    ref_rows = rows_by_role
    source_rows = prior_source_closure
    descriptor = {
        "schema": DESCRIPTOR_SCHEMA,
        "status": DESCRIPTOR_STATUS,
        "qualification": {
            "observationRef": observation_ref,
            "evidenceFiles": evidence_files,
            "outcome": outcome,
            "readerSummary": reader_summary,
        },
        "providerSourceRefs": provider_source_refs,
        "providerToolRefs": provider_tool_refs,
        "approvedPriorArtifactRows": approved_prior_artifact_rows,
        "priorSourceClosure": prior_source_closure,
        "nativeCopyAllowanceSource": native_copy_allowance_source,
        "nativeSharedBudgetSource": native_shared_budget_source,
        "capture": {
            "cohortNetIds": result_trial.get("cohortNetIds"),
            "capturedNetIds": captured.get("netIds"),
            "trialDisposition": result_trial,
            "controlDisposition": control_outcome["disposition"],
            "trialOutcome": trial_outcome["disposition"],
            "passIndex": captured.get("pass"),
            "failedNetId": captured.get("failedNetId"),
            "priorPromotedNetId": captured.get("priorPromotedNetId"),
            "liveSearchIndex": captured.get("liveSearchIndex"),
        },
        "ownerSettings": owner,
        "seedValues": seeds_raw,
        "priorOperationReference": {
            "operation1Sha256": _sha(prior_operation_raw),
            "requestIndex": result_operation.get("requestIndex"),
            "selectedAttempt": result_operation.get("selectedAttempt"),
            "receiptOrdinal": reference.get("receiptOrdinal"),
            "groupOrdinal": reference.get("groupOrdinal"),
            "lineageOrdinal": reference.get("lineageOrdinal"),
            "pass": reference.get("pass"),
            "failedNetId": reference.get("netId"),
            "startPadId": reference.get("startPadId"),
            "rootPadId": reference.get("rootPadId"),
            "reason": reference.get("terminalReason"),
            "reachedNetId": (reference.get("reachedPadIds") or [None])[0],
            "operationBytes": reference.get("prefixGeometryBytes"),
            "holeCount": reference.get("prefixHoleCount"),
            "prefixGeometrySha256": reference.get("prefixGeometrySha256"),
        },
        "sourceIdentity": {
            "geometryBeforeSha256": source.get("templateGeometryBeforeSha256"),
            "geometryAfterSha256": source.get("templateGeometryAfterSha256"),
            "detachedGeometrySha256": source.get("detachedGeometrySha256"),
            "effectsBeforeSha256": source.get("templateEffectsBeforeSha256"),
            "effectsAfterSha256": source.get("templateEffectsAfterSha256"),
            "traceCount": source.get("templateTraceCount"),
            "holeCount": source.get("templateHoleCount"),
            "padLandIdentity": {
                "format": pad_identity.get("encoding"),
                "rowCount": pad_identity.get("padCount"),
                "sha256": pad_identity.get("sha256"),
            },
        },
        "claims": {key: False for key in sorted(CLAIM_KEYS)},
    }
    # Keep the explicit local bindings above visible: these inputs are required,
    # and their role/value shape is checked by the final strict validator.
    _need(type(ref_rows) is dict and type(source_rows) is dict, "DESCRIPTOR_INPUT_BINDINGS")
    return validate_descriptor(descriptor)


def read_descriptor(path, expected_sha256=None):
    candidate = Path(path)
    _need(candidate.is_absolute() and str(candidate) == str(candidate.resolve(strict=True)),
          "DESCRIPTOR_PATH")
    raw = candidate.read_bytes()
    value = validate_descriptor(strict_document(raw))
    digest = _sha(raw)
    if expected_sha256 is not None:
        _need(type(expected_sha256) is str and HEX.fullmatch(expected_sha256) is not None
              and digest == expected_sha256, "DESCRIPTOR_HASH")
    return value, raw, digest


def project_generated_data(descriptor, descriptor_sha256, prior_result):
    descriptor = validate_descriptor(descriptor)
    _hash(descriptor_sha256, "GENERATED_DESCRIPTOR_SHA")
    _need(type(prior_result) is dict and type(prior_result.get("runId")) is str,
          "GENERATED_PRIOR_RESULT")
    qualification = descriptor["qualification"]
    summary = qualification["readerSummary"]
    source = descriptor["priorSourceClosure"]
    operation = descriptor["priorOperationReference"]
    rows = {row["role"]: row for row in descriptor["approvedPriorArtifactRows"]}
    trial = descriptor["capture"]["trialDisposition"]
    _need(summary["resultSha256"] == rows["data-result"]["sha256"]
          and summary["operation1Sha256"] == operation["operation1Sha256"]
          and prior_result.get("operation1", {}).get("receiptSha256")
          == operation["operation1Sha256"], "GENERATED_PRIOR_BINDING")
    qualified = {
        "schema": QUALIFIED_PRIOR_SCHEMA,
        "descriptorSha256": descriptor_sha256,
        "sourceParent": source["sourceParent"],
        "manifestSha256": summary["manifestSha256"],
        "sourceManifestSha256": source["sourceManifestRef"]["sha256"],
        "readerSha256": summary["readerSha256"],
        "resultSha256": summary["resultSha256"],
        "operation1Sha256": operation["operation1Sha256"],
        "parentReceiptSha256": rows["call-parent"]["sha256"],
        "priorRunId": prior_result["runId"],
        "priorOriginalReceipt": {
            "receiptPath": "op-1.jsonl",
            "receiptBytes": rows["data-op-1"]["bytes"],
            "receiptSha256": rows["data-op-1"]["sha256"],
            "receiptOrdinal": operation["receiptOrdinal"],
            "lineageOrdinal": operation["lineageOrdinal"],
        },
        "trialDisposition": {
            key: trial[key] for key in (
                "status", "reason", "capturedFailedNetId", "controlFailedNetId",
                "cohortNetIds", "requestedNetIds", "orderChanged", "feedbackValidated",
            )
        },
    }
    _exact(qualified, QUALIFIED_PRIOR_KEYS, "GENERATED_QUALIFIED_PRIOR_KEYS")
    _exact(qualified["priorOriginalReceipt"], PRIOR_RECEIPT_KEYS,
           "GENERATED_PRIOR_RECEIPT_KEYS")
    _exact(qualified["trialDisposition"], WIRE_TRIAL_DISPOSITION_KEYS,
           "GENERATED_TRIAL_KEYS")
    return {
        "qualifiedPrior": qualified,
        "capture": descriptor["capture"],
        "ownerSettings": descriptor["ownerSettings"],
        "seedValues": descriptor["seedValues"],
        "sourceIdentity": descriptor["sourceIdentity"],
    }


def data_entries(value):
    _need(type(value) in (dict, list), "DATA_ROOT_TYPE")
    total = 1
    if type(value) is dict:
        total += len(value)
        for child in value.values():
            if type(child) in (dict, list):
                total += data_entries(child)
    else:
        total += len(value)
        for child in value:
            if type(child) in (dict, list):
                total += data_entries(child)
    return total


def _load_literal_encoder():
    path = Path(__file__).resolve().parent / "input_literals_r2.py"
    raw = path.read_bytes()
    _need(_sha(raw) == "58ac9b029097d63e078adb70044e31aed29cd5e8d3bea1afee8b54df01a278e9",
          "LITERAL_ENCODER_HASH")
    module = types.ModuleType("_tsj_input_literals_r2")
    module.__file__ = str(path)
    try:
        exec(compile(raw, str(path), "exec"), module.__dict__)
    except Exception:
        raise DescriptorInputError("LITERAL_ENCODER_LOAD") from None
    _need(callable(getattr(module, "java_ascii_literal", None)),
          "LITERAL_ENCODER_API")
    return module.java_ascii_literal


def generate_input_java(descriptor, descriptor_sha256, prior_result):
    tree = project_generated_data(descriptor, descriptor_sha256, prior_result)
    _need(data_entries(tree) <= 4096, "GENERATED_DATA_ENTRIES")
    literal = _load_literal_encoder()
    lines = [
        "package com.lushprojects.circuitjs1.client;",
        "",
        "final class CohortTerminalTelemetryInput {",
        "    static String descriptorSha256() { return " + literal(descriptor_sha256) + "; }",
        "    static int dataEntries() { return " + str(data_entries(tree)) + "; }",
        "    static java.util.Map<String,Object> data() {",
    ]
    serial = [0]

    def emit(value, indent):
        if type(value) is dict:
            name = "m" + str(serial[0])
            serial[0] += 1
            lines.append(indent + "java.util.Map<String,Object> " + name +
                         " = new java.util.LinkedHashMap<String,Object>();")
            for key, child in value.items():
                expression = emit(child, indent)
                lines.append(indent + name + ".put(" + literal(key) + ", " +
                             expression + ");")
            return name
        if type(value) is list:
            name = "l" + str(serial[0])
            serial[0] += 1
            lines.append(indent + "java.util.List<Object> " + name +
                         " = new java.util.ArrayList<Object>();")
            for child in value:
                expression = emit(child, indent)
                lines.append(indent + name + ".add(" + expression + ");")
            return name
        if type(value) is str:
            return literal(value)
        if type(value) is bool:
            return "true" if value else "false"
        if value is None:
            return "null"
        _need(type(value) is int and -(2**31) <= value <= 2**31 - 1,
              "GENERATED_INTEGER")
        return "Integer.valueOf(" + str(value) + ")"

    root_name = emit(tree, "        ")
    lines.append("        return " + root_name + ";")
    lines.append("    }")
    lines.append("}")
    raw = ("\n".join(lines) + "\n").encode("ascii")
    _need(len(raw) <= MAX_GENERATED_BYTES, "GENERATED_SOURCE_SIZE")
    _need(raw.count(b"\\") > 0 and b"java.util.Map<String,Object>" in raw,
          "GENERATED_SOURCE_CONTENT")
    return raw
