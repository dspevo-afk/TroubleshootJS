package com.lushprojects.circuitjs1.client;

/** Independent published9 area budgets; routing, isolation and full terminal geometry. */
public final class PcbOrientationPackingContractTest {
    public static void main(String[] args) {
        long[] seeds={-1017L,-1004L,-1022L,0L,17L,42L};
        long[] areas={4529785,4466664,6943752,4358809,5814368,6211431};
        int turned=0;
        for(int i=0;i<seeds.length;i++) {
            Rb56Plan plan=Rb56Plan.resolve(seeds[i],Rb56Placement.CURRENT_VERSION);
            TroubleshootBoard board=plan.board();
            MediumBoardPhysicalPolicy.Result result=new SeededPcbLayoutGenerator()
                .generateWithPolicyResult(board,plan.layoutSeed,plan.routingSeed);
            System.out.println("PCB_ORIENTATION_ROUTE seed="+seeds[i]+" "+result.toCanonical());
            if(!result.accepted())throw new AssertionError("lost orientation fixture route: "+seeds[i]);
            PcbBoardLayout layout=result.getLayout();
            layout.validateGeometry(board);layout.validateRoutingGeometry(board);layout.validateRouteQuality();
            new PcbTwoLayerRules(board,layout).validate(layout);
            MediumBoardNormalAdmission.fromAcceptedRb56Route(board,result,plan.layoutSeed,plan.routingSeed)
                .requireConstruction(board,layout);
            Rectangle outline=layout.getBoardOutline();
            long area=(long)outline.width*outline.height;
            int rotations=0;
            for(PcbComponentPlacement part:layout.getComponents())
                if(part.getPose().getRotation()==PcbRotation.DEG_180)rotations++;
            turned+=rotations;
            System.out.println("PCB_ORIENTATION_CASE seed="+seeds[i]+" area="+area+" baseline="+areas[i]+
                " turned="+rotations+" geometry="+PlayerSessionFingerprint.of(layout.geometryFingerprint()));
            if(area>areas[i])throw new AssertionError("orientation area regression: "+seeds[i]);
        }
        if(turned==0)throw new AssertionError("no actual declared orientation survived routing");
        System.out.println("PASS: orientation packing contracts roots=6 turned="+turned);
    }
}
